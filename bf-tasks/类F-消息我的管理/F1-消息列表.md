### 1. 修改云函数（comment/index.js）

添加获取用户消息列表的操作：

```javascript
// 获取用户消息列表
async function handleGetMessageList(event, context) {
  const db = cloud.database();
  const _ = db.command;
  const { OPENID } = cloud.getWXContext();
  const { page = 1, pageSize = 20 } = event;
  
  try {
    // 1. 获取用户发布的帖子
    const userPosts = await db.collection('posts')
      .where({
        _openid: OPENID,
        status: 1
      })
      .field({ _id: true })
      .get();
    
    const postIds = userPosts.data.map(post => post._id);
    
    if (postIds.length === 0) {
      return {
        code: 0,
        message: '暂无消息',
        data: []
      };
    }
    
    // 2. 获取这些帖子的评论（排除自己的评论）
    const comments = await db.collection('comments')
      .where({
        postId: _.in(postIds),
        _openid: _.neq(OPENID), // 排除自己的评论
        status: 1
      })
      .orderBy('create_time', 'desc')
      .skip((page - 1) * pageSize)
      .limit(pageSize)
      .get();
    
    // 3. 按帖子聚合评论
    const messageMap = {};
    comments.data.forEach(comment => {
      if (!messageMap[comment.postId]) {
        messageMap[comment.postId] = {
          postId: comment.postId,
          comments: [],
          lastCommentTime: comment.create_time
        };
      }
      messageMap[comment.postId].comments.push(comment);
      // 更新最后评论时间
      if (comment.create_time > messageMap[comment.postId].lastCommentTime) {
        messageMap[comment.postId].lastCommentTime = comment.create_time;
      }
    });
    
    // 4. 获取帖子信息
    const messageList = await Promise.all(
      Object.values(messageMap).map(async (message) => {
        const post = await db.collection('posts')
          .doc(message.postId)
          .field({ title: true, content: true })
          .get();
        
        return {
          ...message,
          postTitle: post.data.title,
          postContent: post.data.content,
          commentCount: message.comments.length
        };
      })
    );
    
    // 5. 按最后评论时间排序
    messageList.sort((a, b) => new Date(b.lastCommentTime) - new Date(a.lastCommentTime));
    
    return {
      code: 0,
      message: '获取消息列表成功',
      data: messageList
    };
  } catch (err) {
    return {
      code: -1,
      message: '获取消息列表失败',
      error: err.message
    };
  }
}
```

### 2. 创建消息页面（pages/message/message）

#### message.js

```javascript
const app = getApp();

Page({
  data: {
    messageList: [],
    page: 1,
    pageSize: 20,
    hasMore: true,
    loading: false
  },
  
  onLoad() {
    this.loadMessages();
  },
  
  onShow() {
    // 每次显示页面时刷新消息
    this.setData({ page: 1, messageList: [] });
    this.loadMessages();
  },
  
  // 加载消息列表
  async loadMessages() {
    if (this.data.loading || !this.data.hasMore) return;
    
    this.setData({ loading: true });
    
    try {
      const res = await wx.cloud.callFunction({
        name: 'comment',
        data: {
          action: 'messageList',
          page: this.data.page,
          pageSize: this.data.pageSize
        }
      });
      
      if (res.result.code === 0) {
        const newMessages = res.result.data;
        this.setData({
          messageList: this.data.page === 1 ? newMessages : [...this.data.messageList, ...newMessages],
          hasMore: newMessages.length === this.data.pageSize,
          page: this.data.page + 1
        });
      }
    } catch (err) {
      console.error('获取消息失败', err);
      wx.showToast({
        title: '获取消息失败',
        icon: 'none'
      });
    } finally {
      this.setData({ loading: false });
    }
  },
  
  // 下拉刷新
  onPullDownRefresh() {
    this.setData({ page: 1, messageList: [], hasMore: true });
    this.loadMessages().then(() => {
      wx.stopPullDownRefresh();
    });
  },
  
  // 上拉加载更多
  onReachBottom() {
    this.loadMessages();
  },
  
  // 点击消息跳转到帖子详情
  onMessageTap(e) {
    const postId = e.currentTarget.dataset.postId;
    wx.navigateTo({
      url: `/pages/post-detail/post-detail?id=${postId}`
    });
  }
});
```

## 数据模型

### comments集合（已有字段）
- `_id` - 评论ID
- `postId` - 帖子ID
- `_openid` - 评论者openid
- `content` - 评论内容
- `create_time` - 创建时间
- `status` - 状态（1=正常）

### posts集合（已有字段）
- `_id` - 帖子ID
- `_openid` - 作者openid
- `title` - 帖子标题
- `content` - 帖子内容

## 验收标准

- ✅ 别人评论我的帖 → 消息页能看到
- ✅ 消息按帖子聚合展示
- ✅ 点击消息可跳转到帖子详情
- ✅ 支持下拉刷新和上拉加载更多

## 测试方法

1. 用户A发布帖子
2. 用户B评论该帖子
3. 用户A进入消息页
4. 验证消息显示正确
5. 点击消息跳转到帖子详情

## 注意事项

1. 只显示别人对自己帖子的评论，不显示自己的评论
2. 按帖子聚合，每个帖子显示最新评论
3. 消息按最后评论时间倒序排列
4. 支持分页加载
