## 实现步骤

### 1. 创建收藏云函数（cloudfunctions/favorite）

#### favorite/index.js

```javascript
const cloud = require('wx-server-sdk');
cloud.init({
  env: cloud.DYNAMIC_CURRENT_ENV
});
const db = cloud.database();
const _ = db.command;

exports.main = async (event, context) => {
  const { OPENID } = cloud.getWXContext();
  const { action, postId } = event;
  
  switch (action) {
    case 'add':
      return await handleAddFavorite(event, context, OPENID);
    case 'remove':
      return await handleRemoveFavorite(event, context, OPENID);
    case 'check':
      return await handleCheckFavorite(event, context, OPENID);
    case 'list':
      return await handleGetFavoriteList(event, context, OPENID);
    default:
      return {
        code: -1,
        message: '未知操作'
      };
  }
};

// 添加收藏
async function handleAddFavorite(event, context, openid) {
  const { postId } = event;
  
  try {
    // 检查是否已收藏
    const existing = await db.collection('favorites')
      .where({
        _openid: openid,
        postId: postId
      })
      .get();
    
    if (existing.data.length > 0) {
      return {
        code: -1,
        message: '已收藏该帖子'
      };
    }
    
    // 添加收藏
    const result = await db.collection('favorites')
      .add({
        data: {
          _openid: openid,
          postId: postId,
          create_time: db.serverDate()
        }
      });
    
    return {
      code: 0,
      message: '收藏成功',
      data: result
    };
  } catch (err) {
    return {
      code: -1,
      message: '收藏失败',
      error: err.message
    };
  }
}

// 取消收藏
async function handleRemoveFavorite(event, context, openid) {
  const { postId } = event;
  
  try {
    const result = await db.collection('favorites')
      .where({
        _openid: openid,
        postId: postId
      })
      .remove();
    
    return {
      code: 0,
      message: '取消收藏成功',
      data: result
    };
  } catch (err) {
    return {
      code: -1,
      message: '取消收藏失败',
      error: err.message
    };
  }
}

// 检查是否已收藏
async function handleCheckFavorite(event, context, openid) {
  const { postId } = event;
  
  try {
    const result = await db.collection('favorites')
      .where({
        _openid: openid,
        postId: postId
      })
      .get();
    
    return {
      code: 0,
      data: {
        isFavorite: result.data.length > 0
      }
    };
  } catch (err) {
    return {
      code: -1,
      message: '检查收藏状态失败',
      error: err.message
    };
  }
}

// 获取收藏列表
async function handleGetFavoriteList(event, context, openid) {
  const { page = 1, pageSize = 20 } = event;
  
  try {
    // 获取收藏记录
    const favorites = await db.collection('favorites')
      .where({
        _openid: openid
      })
      .orderBy('create_time', 'desc')
      .skip((page - 1) * pageSize)
      .limit(pageSize)
      .get();
    
    // 获取帖子信息
    const posts = await Promise.all(
      favorites.data.map(async (favorite) => {
        const post = await db.collection('posts')
          .doc(favorite.postId)
          .get();
        
        return {
          ...post.data,
          favoriteTime: favorite.create_time
        };
      })
    );
    
    return {
      code: 0,
      message: '获取收藏列表成功',
      data: posts
    };
  } catch (err) {
    return {
      code: -1,
      message: '获取收藏列表失败',
      error: err.message
    };
  }
}
```

### 2. 修改帖子详情页（pages/post-detail/post-detail.js）

添加收藏功能：

```javascript
Page({
  data: {
    post: null,
    isFavorite: false,
    favoriteLoading: false
  },
  
  onLoad(options) {
    this.setData({ postId: options.id });
    this.loadPostDetail();
    this.checkFavoriteStatus();
  },
  
  // 检查收藏状态
  async checkFavoriteStatus() {
    try {
      const res = await wx.cloud.callFunction({
        name: 'favorite',
        data: {
          action: 'check',
          postId: this.data.postId
        }
      });
      
      if (res.result.code === 0) {
        this.setData({
          isFavorite: res.result.data.isFavorite
        });
      }
    } catch (err) {
      console.error('检查收藏状态失败', err);
    }
  },
  
  // 收藏/取消收藏
  async onFavoriteTap() {
    if (this.data.favoriteLoading) return;
    
    this.setData({ favoriteLoading: true });
    
    try {
      const action = this.data.isFavorite ? 'remove' : 'add';
      const res = await wx.cloud.callFunction({
        name: 'favorite',
        data: {
          action,
          postId: this.data.postId
        }
      });
      
      if (res.result.code === 0) {
        this.setData({
          isFavorite: !this.data.isFavorite
        });
        
        wx.showToast({
          title: this.data.isFavorite ? '收藏成功' : '取消收藏成功',
          icon: 'success'
        });
      } else {
        wx.showToast({
          title: res.result.message,
          icon: 'none'
        });
      }
    } catch (err) {
      console.error('收藏操作失败', err);
      wx.showToast({
        title: '操作失败',
        icon: 'none'
      });
    } finally {
      this.setData({ favoriteLoading: false });
    }
  }
});
```

### 5. 创建收藏列表页面（pages/my-favorites/my-favorites）

#### my-favorites.js

```javascript
Page({
  data: {
    favorites: [],
    page: 1,
    pageSize: 20,
    hasMore: true,
    loading: false
  },
  
  onLoad() {
    this.loadFavorites();
  },
  
  async loadFavorites() {
    if (this.data.loading || !this.data.hasMore) return;
    
    this.setData({ loading: true });
    
    try {
      const res = await wx.cloud.callFunction({
        name: 'favorite',
        data: {
          action: 'list',
          page: this.data.page,
          pageSize: this.data.pageSize
        }
      });
      
      if (res.result.code === 0) {
        const newFavorites = res.result.data;
        this.setData({
          favorites: this.data.page === 1 ? newFavorites : [...this.data.favorites, ...newFavorites],
          hasMore: newFavorites.length === this.data.pageSize,
          page: this.data.page + 1
        });
      }
    } catch (err) {
      console.error('获取收藏列表失败', err);
    } finally {
      this.setData({ loading: false });
    }
  },
  
  onPullDownRefresh() {
    this.setData({ page: 1, favorites: [], hasMore: true });
    this.loadFavorites().then(() => {
      wx.stopPullDownRefresh();
    });
  },
  
  onReachBottom() {
    this.loadFavorites();
  },
  
  onFavoriteTap(e) {
    const postId = e.currentTarget.dataset.postId;
    wx.navigateTo({
      url: `/pages/post-detail/post-detail?id=${postId}`
    });
  },
  
  // 取消收藏
  async onRemoveFavorite(e) {
    const postId = e.currentTarget.dataset.postId;
    
    wx.showModal({
      title: '提示',
      content: '确定要取消收藏吗？',
      success: async (res) => {
        if (res.confirm) {
          try {
            const result = await wx.cloud.callFunction({
              name: 'favorite',
              data: {
                action: 'remove',
                postId
              }
            });
            
            if (result.result.code === 0) {
              // 从列表中移除
              const favorites = this.data.favorites.filter(fav => fav._id !== postId);
              this.setData({ favorites });
              
              wx.showToast({
                title: '取消收藏成功',
                icon: 'success'
              });
            }
          } catch (err) {
            console.error('取消收藏失败', err);
            wx.showToast({
              title: '操作失败',
              icon: 'none'
            });
          }
        }
      }
    });
  }
});
```
