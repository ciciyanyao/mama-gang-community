```javascript
const app = getApp();

Page({
  data: {
    userInfo: null,
    hasUserInfo: false,
    canIUse: wx.canIUse('button.open-type.getUserInfo'),
    menuList: [
      {
        id: 'my-posts',
        title: '我的帖子',
        icon: '/images/post-icon.svg',
        url: '/pages/my-posts/my-posts'
      },
      {
        id: 'my-favorites',
        title: '我的收藏',
        icon: '/images/favorite-icon.svg',
        url: '/pages/my-favorites/my-favorites'
      },
      {
        id: 'settings',
        title: '设置',
        icon: '/images/setting-icon.svg',
        url: '/pages/settings/settings'
      },
      {
        id: 'about',
        title: '关于',
        icon: '/images/about-icon.svg',
        url: '/pages/about/about'
      }
    ]
  },
  
  onLoad() {
    if (app.globalData.userInfo) {
      this.setData({
        userInfo: app.globalData.userInfo,
        hasUserInfo: true
      });
    } else if (this.data.canIUse) {
      // 由于 getUserInfo 是网络请求，可能会在 Page.onLoad 之后才返回
      // 所以此处加入 callback 以防止这种情况
      app.userInfoReadyCallback = res => {
        this.setData({
          userInfo: res.userInfo,
          hasUserInfo: true
        });
      };
    } else {
      // 在没有 open-type=getUserInfo 版本的兼容处理
      wx.getUserInfo({
        success: res => {
          app.globalData.userInfo = res.userInfo;
          this.setData({
            userInfo: res.userInfo,
            hasUserInfo: true
          });
        }
      });
    }
  },
  
  onShow() {
    // 每次显示页面时刷新用户信息
    this.refreshUserInfo();
  },
  
  // 刷新用户信息
  async refreshUserInfo() {
    try {
      const res = await wx.cloud.callFunction({
        name: 'user',
        data: {
          action: 'getUserInfo'
        }
      });
      
      if (res.result.code === 0) {
        const userInfo = res.result.data;
        app.globalData.userInfo = userInfo;
        this.setData({
          userInfo,
          hasUserInfo: true
        });
      }
    } catch (err) {
      console.error('获取用户信息失败', err);
    }
  },
  
  // 获取用户信息
  getUserInfo(e) {
    if (e.detail.userInfo) {
      app.globalData.userInfo = e.detail.userInfo;
      this.setData({
        userInfo: e.detail.userInfo,
        hasUserInfo: true
      });
    }
  },
  
  // 跳转到菜单页面
  onMenuTap(e) {
    const url = e.currentTarget.dataset.url;
    wx.navigateTo({
      url
    });
  },
  
  // 退出登录
  onLogout() {
    wx.showModal({
      title: '提示',
      content: '确定要退出登录吗？',
      success: (res) => {
        if (res.confirm) {
          // 清除用户信息
          app.globalData.userInfo = null;
          this.setData({
            userInfo: null,
            hasUserInfo: false
          });
          
          // 跳转到登录页
          wx.redirectTo({
            url: '/pages/login/login'
          });
        }
      }
    });
  }
});
```

### 2. 创建子页面

#### 我的帖子页面（pages/my-posts/my-posts）

```javascript
// my-posts.js
Page({
  data: {
    posts: [],
    page: 1,
    pageSize: 20,
    hasMore: true,
    loading: false
  },
  
  onLoad() {
    this.loadPosts();
  },
  
  async loadPosts() {
    if (this.data.loading || !this.data.hasMore) return;
    
    this.setData({ loading: true });
    
    try {
      const res = await wx.cloud.callFunction({
        name: 'post',
        data: {
          action: 'myPosts',
          page: this.data.page,
          pageSize: this.data.pageSize
        }
      });
      
      if (res.result.code === 0) {
        const newPosts = res.result.data;
        this.setData({
          posts: this.data.page === 1 ? newPosts : [...this.data.posts, ...newPosts],
          hasMore: newPosts.length === this.data.pageSize,
          page: this.data.page + 1
        });
      }
    } catch (err) {
      console.error('获取我的帖子失败', err);
    } finally {
      this.setData({ loading: false });
    }
  },
  
  onPullDownRefresh() {
    this.setData({ page: 1, posts: [], hasMore: true });
    this.loadPosts().then(() => {
      wx.stopPullDownRefresh();
    });
  },
  
  onReachBottom() {
    this.loadPosts();
  },
  
  onPostTap(e) {
    const postId = e.currentTarget.dataset.postId;
    wx.navigateTo({
      url: `/pages/post-detail/post-detail?id=${postId}`
    });
  }
});
```

#### 设置页面（pages/settings/settings）

```javascript
// settings.js
Page({
  data: {
    userInfo: null,
    settings: {
      notification: true,
      privacy: false
    }
  },
  
  onLoad() {
    this.loadUserInfo();
    this.loadSettings();
  },
  
  async loadUserInfo() {
    try {
      const res = await wx.cloud.callFunction({
        name: 'user',
        data: {
          action: 'getUserInfo'
        }
      });
      
      if (res.result.code === 0) {
        this.setData({
          userInfo: res.result.data
        });
      }
    } catch (err) {
      console.error('获取用户信息失败', err);
    }
  },
  
  loadSettings() {
    // 从本地存储加载设置
    const settings = wx.getStorageSync('userSettings') || this.data.settings;
    this.setData({ settings });
  },
  
  onNotificationChange(e) {
    const settings = this.data.settings;
    settings.notification = e.detail.value;
    this.setData({ settings });
    wx.setStorageSync('userSettings', settings);
  },
  
  onPrivacyChange(e) {
    const settings = this.data.settings;
    settings.privacy = e.detail.value;
    this.setData({ settings });
    wx.setStorageSync('userSettings', settings);
  },
  
  onAboutTap() {
    wx.navigateTo({
      url: '/pages/about/about'
    });
  },
  
  onClearCache() {
    wx.showModal({
      title: '提示',
      content: '确定要清除缓存吗？',
      success: (res) => {
        if (res.confirm) {
          wx.clearStorageSync();
          wx.showToast({
            title: '清除成功',
            icon: 'success'
          });
        }
      }
    });
  }
});
```
