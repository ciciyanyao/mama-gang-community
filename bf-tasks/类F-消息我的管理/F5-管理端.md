## 任务说明

- **优先级**：P0
- **目标**：管理员管理社区
- **功能**：删帖、删评论、处理举报
- **权限**：users.role=2的用户
- **依赖**：C的举报功能

## 实现步骤

### 1. 创建管理云函数（cloudfunctions/admin）

#### admin/index.js

```javascript
const cloud = require('wx-server-sdk');
cloud.init({
  env: cloud.DYNAMIC_CURRENT_ENV
});
const db = cloud.database();
const _ = db.command;

exports.main = async (event, context) => {
  const { OPENID } = cloud.getWXContext();
  const { action } = event;
  
  // 验证管理员权限
  const isAdmin = await checkAdminRole(OPENID);
  if (!isAdmin) {
    return {
      code: -1,
      message: '权限不足，需要管理员权限'
    };
  }
  
  switch (action) {
    case 'deletePost':
      return await handleDeletePost(event, context);
    case 'deleteComment':
      return await handleDeleteComment(event, context);
    case 'handleReport':
      return await handleReport(event, context);
    case 'getReportList':
      return await handleGetReportList(event, context);
    case 'setTop':
      return await handleSetTop(event, context);
    case 'getAdminStats':
      return await handleGetAdminStats(event, context);
    default:
      return {
        code: -1,
        message: '未知操作'
      };
  }
};

// 检查管理员权限
async function checkAdminRole(openid) {
  try {
    const user = await db.collection('users')
      .where({
        _openid: openid,
        role: 2 // 管理员角色
      })
      .get();
    
    return user.data.length > 0;
  } catch (err) {
    console.error('检查管理员权限失败', err);
    return false;
  }
}

// 删除帖子
async function handleDeletePost(event, context) {
  const { postId, reason } = event;
  
  try {
    // 更新帖子状态为已删除
    const result = await db.collection('posts')
      .doc(postId)
      .update({
        data: {
          status: 2, // 已删除
          delete_reason: reason || '管理员删除',
          delete_time: db.serverDate()
        }
      });
    
    // 同时删除该帖子下的所有评论
    await db.collection('comments')
      .where({
        postId: postId
      })
      .update({
        data: {
          status: 2,
          delete_reason: '帖子已被删除',
          delete_time: db.serverDate()
        }
      });
    
    return {
      code: 0,
      message: '删除帖子成功',
      data: result
    };
  } catch (err) {
    return {
      code: -1,
      message: '删除帖子失败',
      error: err.message
    };
  }
}

// 删除评论
async function handleDeleteComment(event, context) {
  const { commentId, reason } = event;
  
  try {
    const result = await db.collection('comments')
      .doc(commentId)
      .update({
        data: {
          status: 2, // 已删除
          delete_reason: reason || '管理员删除',
          delete_time: db.serverDate()
        }
      });
    
    // 更新帖子评论数
    const comment = await db.collection('comments')
      .doc(commentId)
      .get();
    
    if (comment.data) {
      await db.collection('posts')
        .doc(comment.data.postId)
        .update({
          data: {
            commentCount: _.inc(-1)
          }
        });
    }
    
    return {
      code: 0,
      message: '删除评论成功',
      data: result
    };
  } catch (err) {
    return {
      code: -1,
      message: '删除评论失败',
      error: err.message
    };
  }
}

// 处理举报
async function handleReport(event, context) {
  const { reportId, action, reason } = event;
  
  try {
    // 获取举报信息
    const report = await db.collection('reports')
      .doc(reportId)
      .get();
    
    if (!report.data) {
      return {
        code: -1,
        message: '举报记录不存在'
      };
    }
    
    // 根据处理动作执行相应操作
    switch (action) {
      case 'approve':
        // 批准举报，删除被举报内容
        if (report.data.targetType === 'post') {
          await handleDeletePost({
            postId: report.data.targetId,
            reason: '举报批准：' + (reason || report.data.reason)
          }, context);
        } else if (report.data.targetType === 'comment') {
          await handleDeleteComment({
            commentId: report.data.targetId,
            reason: '举报批准：' + (reason || report.data.reason)
          }, context);
        }
        
        // 更新举报状态
        await db.collection('reports')
          .doc(reportId)
          .update({
            data: {
              status: 2, // 已处理
              handle_time: db.serverDate(),
              handle_result: 'approved'
            }
          });
        
        return {
          code: 0,
          message: '举报已批准，内容已删除'
        };
      
      case 'reject':
        // 驳回举报
        await db.collection('reports')
          .doc(reportId)
          .update({
            data: {
              status: 3, // 已驳回
              handle_time: db.serverDate(),
              handle_result: 'rejected',
              reject_reason: reason || '举报内容不符合删除条件'
            }
          });
        
        return {
          code: 0,
          message: '举报已驳回'
        };
      
      default:
        return {
          code: -1,
          message: '未知处理动作'
        };
    }
  } catch (err) {
    return {
      code: -1,
      message: '处理举报失败',
      error: err.message
    };
  }
}

// 获取举报列表
async function handleGetReportList(event, context) {
  const { page = 1, pageSize = 20, status } = event;
  
  try {
    let query = db.collection('reports');
    
    // 按状态筛选
    if (status) {
      query = query.where({
        status: parseInt(status)
      });
    }
    
    const result = await query
      .orderBy('create_time', 'desc')
      .skip((page - 1) * pageSize)
      .limit(pageSize)
      .get();
    
    // 获取被举报内容信息
    const reports = await Promise.all(
      result.data.map(async (report) => {
        let targetInfo = null;
        
        try {
          if (report.targetType === 'post') {
            const post = await db.collection('posts')
              .doc(report.targetId)
              .field({ title: true, content: true })
              .get();
            targetInfo = post.data;
          } else if (report.targetType === 'comment') {
            const comment = await db.collection('comments')
              .doc(report.targetId)
              .field({ content: true })
              .get();
            targetInfo = comment.data;
          }
        } catch (err) {
          console.error('获取被举报内容失败', err);
        }
        
        return {
          ...report,
          targetInfo
        };
      })
    );
    
    return {
      code: 0,
      message: '获取举报列表成功',
      data: reports
    };
  } catch (err) {
    return {
      code: -1,
      message: '获取举报列表失败',
      error: err.message
    };
  }
}

// 设置置顶
async function handleSetTop(event, context) {
  const { postId, isTop } = event;
  
  try {
    const result = await db.collection('posts')
      .doc(postId)
      .update({
        data: {
          is_top: isTop
        }
      });
    
    return {
      code: 0,
      message: isTop ? '帖子已置顶' : '帖子已取消置顶',
      data: result
    };
  } catch (err) {
    return {
      code: -1,
      message: '设置置顶失败',
      error: err.message
    };
  }
}

// 获取管理统计数据
async function handleGetAdminStats(event, context) {
  try {
    // 获取今日新增帖子数
    const today = new Date();
    today.setHours(0, 0, 0, 0);
    
    const todayPosts = await db.collection('posts')
      .where({
        create_time: _.gte(today),
        status: 1
      })
      .count();
    
    // 获取今日新增评论数
    const todayComments = await db.collection('comments')
      .where({
        create_time: _.gte(today),
        status: 1
      })
      .count();
    
    // 获取待处理举报数
    const pendingReports = await db.collection('reports')
      .where({
        status: 1
      })
      .count();
    
    // 获取总用户数
    const totalUsers = await db.collection('users')
      .count();
    
    return {
      code: 0,
      data: {
        todayPosts: todayPosts.total,
        todayComments: todayComments.total,
        pendingReports: pendingReports.total,
        totalUsers: totalUsers.total
      }
    };
  } catch (err) {
    return {
      code: -1,
      message: '获取统计数据失败',
      error: err.message
    };
  }
}
```

```javascript
Page({
  data: {
    stats: {
      todayPosts: 0,
      todayComments: 0,
      pendingReports: 0,
      totalUsers: 0
    },
    reports: [],
    page: 1,
    pageSize: 20,
    hasMore: true,
    loading: false,
    currentTab: 'stats' // stats, reports
  },
  
  onLoad() {
    this.loadStats();
    this.loadReports();
  },
  
  onShow() {
    this.loadStats();
  },
  
  // 加载统计数据
  async loadStats() {
    try {
      const res = await wx.cloud.callFunction({
        name: 'admin',
        data: {
          action: 'getAdminStats'
        }
      });
      
      if (res.result.code === 0) {
        this.setData({
          stats: res.result.data
        });
      }
    } catch (err) {
      console.error('获取统计数据失败', err);
    }
  },
  
  // 加载举报列表
  async loadReports() {
    if (this.data.loading || !this.data.hasMore) return;
    
    this.setData({ loading: true });
    
    try {
      const res = await wx.cloud.callFunction({
        name: 'admin',
        data: {
          action: 'getReportList',
          page: this.data.page,
          pageSize: this.data.pageSize,
          status: 1 // 只加载待处理的
        }
      });
      
      if (res.result.code === 0) {
        const newReports = res.result.data;
        this.setData({
          reports: this.data.page === 1 ? newReports : [...this.data.reports, ...newReports],
          hasMore: newReports.length === this.data.pageSize,
          page: this.data.page + 1
        });
      }
    } catch (err) {
      console.error('获取举报列表失败', err);
    } finally {
      this.setData({ loading: false });
    }
  },
  
  // 切换标签
  onTabChange(e) {
    const tab = e.currentTarget.dataset.tab;
    this.setData({
      currentTab: tab
    });
    
    if (tab === 'reports' && this.data.reports.length === 0) {
      this.loadReports();
    }
  },
  
  // 处理举报
  async onHandleReport(e) {
    const { reportId, action } = e.currentTarget.dataset;
    
    wx.showModal({
      title: '确认处理',
      content: action === 'approve' ? '确定要批准举报并删除内容吗？' : '确定要驳回举报吗？',
      success: async (res) => {
        if (res.confirm) {
          try {
            const result = await wx.cloud.callFunction({
              name: 'admin',
              data: {
                action: 'handleReport',
                reportId,
                action
              }
            });
            
            if (result.result.code === 0) {
              wx.showToast({
                title: result.result.message,
                icon: 'success'
              });
              
              // 刷新列表
              this.setData({
                reports: [],
                page: 1,
                hasMore: true
              });
              this.loadReports();
              this.loadStats();
            } else {
              wx.showToast({
                title: result.result.message,
                icon: 'none'
              });
            }
          } catch (err) {
            console.error('处理举报失败', err);
            wx.showToast({
              title: '处理失败',
              icon: 'none'
            });
          }
        }
      }
    });
  },
  
  // 删除帖子
  async onDeletePost(e) {
    const postId = e.currentTarget.dataset.postId;
    
    wx.showModal({
      title: '确认删除',
      content: '确定要删除这个帖子吗？删除后无法恢复。',
      success: async (res) => {
        if (res.confirm) {
          try {
            const result = await wx.cloud.callFunction({
              name: 'admin',
              data: {
                action: 'deletePost',
                postId,
                reason: '管理员删除'
              }
            });
            
            if (result.result.code === 0) {
              wx.showToast({
                title: '删除成功',
                icon: 'success'
              });
              
              // 刷新列表
              this.setData({
                reports: [],
                page: 1,
                hasMore: true
              });
              this.loadReports();
              this.loadStats();
            }
          } catch (err) {
            console.error('删除帖子失败', err);
            wx.showToast({
              title: '删除失败',
              icon: 'none'
            });
          }
        }
      }
    });
  },
  
  // 删除评论
  async onDeleteComment(e) {
    const commentId = e.currentTarget.dataset.commentId;
    
    wx.showModal({
      title: '确认删除',
      content: '确定要删除这条评论吗？',
      success: async (res) => {
        if (res.confirm) {
          try {
            const result = await wx.cloud.callFunction({
              name: 'admin',
              data: {
                action: 'deleteComment',
                commentId,
                reason: '管理员删除'
              }
            });
            
            if (result.result.code === 0) {
              wx.showToast({
                title: '删除成功',
                icon: 'success'
              });
            }
          } catch (err) {
            console.error('删除评论失败', err);
            wx.showToast({
              title: '删除失败',
              icon: 'none'
            });
          }
        }
      }
    });
  }
});
```

## 数据模型

### reports集合（已有字段）
- `_id` - 举报ID
- `targetId` - 被举报内容ID
- `targetType` - 被举报类型（post/comment）
- `reporterId` - 举报者openid
- `reason` - 举报原因
- `status` - 状态（1=待处理, 2=已处理, 3=已驳回）
- `create_time` - 创建时间

### users集合（新增字段）
- `role` - 用户角色（0=普通用户, 1=版主, 2=管理员）

## 验收标准

- ✅ 管理员可以删除违规帖子
- ✅ 管理员可以删除违规评论
- ✅ 管理员可以处理举报
- ✅ 管理员可以查看统计数据

## 测试方法

1. 使用管理员账号登录
2. 进入管理后台
3. 查看统计数据
4. 处理举报（批准/驳回）
5. 删除违规帖子/评论

## 注意事项

1. 管理员权限验证要严格
2. 删除操作需要二次确认
3. 删除帖子会同时删除该帖子下的所有评论
4. 举报处理结果会更新举报状态
